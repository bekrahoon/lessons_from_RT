💻 Пример кода функций

def display_word(word, guessed_letters):
    """Показывает слово с угаданными буквами"""
    display = ""
    for letter in word:
        if letter in guessed_letters:
            display += letter + " "
        else:
            display += "_ "
    return display.strip()
